Connect a live Google Sheet

The prediction engine can read straight from a Sheet instead of demo data. Deploy the Apps Script below as a Web App, paste its URL here, and each room's risk score recalculates from your real maintenance log.

1. Sheet columns — create a sheet named Assets with this header row:

IDRoomBlockType InstallDateLastMaintenanceComplaints30d Complaints90dUsageIntensity
2. Apps Script backend (copy into Extensions → Apps Script, deploy as Web App) 
function doGet(e) {
  var action = e.parameter.action || 'list';
  if (action === 'list') return listAssets();
  return jsonOut({error: 'unknown action'});
}

function doPost(e) {
  var body = JSON.parse(e.postData.contents);
  if (body.action === 'logMaintenance') return logMaintenance(body.id, body.date);
  return jsonOut({error: 'unknown action'});
}

function getSheet() {
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Assets');
}

function listAssets() {
  var sheet = getSheet();
  var rows = sheet.getDataRange().getValues();
  var headers = rows.shift();
  var out = rows.map(function(r) {
    var obj = {};
    headers.forEach(function(h, i) { obj[h] = r[i]; });
    return obj;
  });
  return jsonOut(out);
}

function logMaintenance(id, dateStr) {
  var sheet = getSheet();
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var idCol = headers.indexOf('ID');
  var dateCol = headers.indexOf('LastMaintenance');
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(id)) {
      sheet.getRange(i + 1, dateCol + 1).setValue(dateStr);
      break;
    }
  }
  return jsonOut({ok: true});
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

Deploy → Execute as: Me, Who has access: Anyone. Paste the resulting /exec URL above.
